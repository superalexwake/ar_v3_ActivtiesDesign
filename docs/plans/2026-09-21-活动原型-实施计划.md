# Activities 活动原型实施计划

> **执行方式：** 推荐使用 superpowers:subagent-driven-development（每个任务派一个子代理，任务之间由主代理审查），也可以用 superpowers:executing-plans 在当前会话逐项执行。步骤用复选框（`- [ ]`）跟踪。

**目标：** 把 ar_v2_vue（`masterBranch/main-3.10`）派生为脱离后端的活动可点击原型：21 种活动一键可达、PC 演示控制台、产品独立 HTML 原型挂载、纯静态构建。

**架构：** 原型代码全部放在 `entrance/prototype`。入口首行导入 mock 层，mock 层把 `axios.defaults.adapter` 换成按接口路径分发的假数据 adapter，业务代码不改。`public/console.html` 通过同源 iframe 驱动 H5。构建产物是纯静态文件，使用相对路径。

**技术栈：** Vue 3.3、Vite 7、Vant 4、Pinia、vue-router 4（hash）、axios 1、dayjs、pnpm 11、Node 22。

**设计文档：** `docs/specs/2026-09-21-活动原型-设计方案.md`（执行者必须同时阅读）

## 全局约束

- 只在本地孤儿分支 `main` 上提交；禁止向任何远端推送，禁止 `git push --all` 和 `git push --mirror`。
- `src/` 下只允许修改 `src/plugins/html.ts`；其余原型代码全部放在 `entrance/prototype/`。
- mock 层只允许导入第三方库（`axios`、`dayjs`）、mock 自身文件、`@/api/url` 和 `import type`。
- 所有接口域名为 `https://prototype.invalid`；假数据只用虚构的用户名、ID 和金额；界面中文，币种 ₹。
- 活动开关字段返回字符串 `'1'` / `'0'`（`useActive.ts` 的 `translateBoolean` 只认 `'1'`）。
- 导出的 TypeScript 声明写中文 TSDoc，标签用 `@param name - 描述`、`@returns`、`@typeParam`、`@remarks`；注释不写变更原因和 TODO。
- 代码风格：早返回，嵌套不超过 3 层，优先用映射表；不新增依赖。
- 按用户的前端测试规范，不写单元测试。验证方式是 `pnpm build` 加上浏览器走查（设计文档 §10）。
- 开发端口 5190。启动 dev server 前先执行 `lsof -ti tcp:5190`，有旧进程先关掉。
- 提交信息格式：`<type>(prototype): <中文摘要>`，结尾附 `Co-Authored-By: Claude Opus 5 <noreply@anthropic.com>`。

import type { MockEnvelope } from './types'

/**
 * 分页列表数据，字段与线上分页接口一致。
 *
 * @typeParam T - 列表项类型。
 */
export interface PagedData<T> {
	/** 当前页数据 */
	list: T[]
	/** 总条数 */
	totalCount: number
	/** 总页数，最小为 1 */
	totalPage: number
	/** 当前页码，从 1 开始 */
	pageNo: number
}

/**
 * 构造成功信封。
 *
 * @typeParam T - 业务数据类型。
 * @param data - 业务数据。
 * @returns `code` 为 0 的信封。
 */
export const ok = <T>(data: T): MockEnvelope<T> => ({ code: 0, msg: 'Succeed', data })

/**
 * 构造业务失败信封；现有响应拦截器会 reject 并提示 `msg`。
 *
 * @param msg - 展示给用户的中文提示。
 * @returns `code` 为 -2 的信封。
 */
export const fail = (msg: string): MockEnvelope<null> => ({ code: -2, msg, data: null })

/**
 * 按请求中的 `pageNo`、`pageSize` 对完整列表分页。
 *
 * @typeParam T - 列表项类型。
 * @param all - 完整列表。
 * @param body - 请求参数；缺少分页参数时按第 1 页、每页 20 条处理。
 * @returns 当前页的分页数据。
 */
export function paged<T>(all: T[], body: Record<string, unknown>): PagedData<T> {
	const pageNo = Number(body.pageNo) || 1
	const pageSize = Number(body.pageSize) || 20
	const start = (pageNo - 1) * pageSize
	return {
		list: all.slice(start, start + pageSize),
		totalCount: all.length,
		totalPage: Math.max(1, Math.ceil(all.length / pageSize)),
		pageNo,
	}
}

import type { ActivitySwitch, PopupKey, RewardState, Scenario } from './types'

/** 场景在 localStorage 中的键，控制台使用同一个键 */
export const SCENARIO_KEY = 'proto:scenario'

const SWITCHES: ActivitySwitch[] = ['activityAward', 'task', 'washCode', 'jackpot', 'memberGift', 'champion', 'invitedWheel', 'turntable']
const REWARD_STATES: RewardState[] = ['claimable', 'claimed', 'expired', 'empty']
const POPUPS: PopupKey[] = ['first-recharge', 'return-award', 'app-download-reward', 're-benefit']

/** 默认场景：已登录、奖励可领取、全部开关打开、不弹窗、延迟 200 毫秒 */
export const DEFAULT_SCENARIO: Scenario = {
	loggedIn: true,
	rewardState: 'claimable',
	switches: Object.fromEntries(SWITCHES.map((key) => [key, true])) as Record<ActivitySwitch, boolean>,
	popup: null,
	latencyMs: 200,
}

/**
 * 读取并校验场景。
 *
 * @remarks localStorage 中的内容可能被手工或 AI 改坏；任一字段不合法时整体回退默认场景并打印警告。
 * @returns 合法的场景。
 */
export function readScenario(): Scenario {
	const raw = localStorage.getItem(SCENARIO_KEY)
	if (!raw) return DEFAULT_SCENARIO
	try {
		const value: unknown = JSON.parse(raw)
		if (isScenario(value)) return value
	} catch {
		// JSON 损坏与字段不合法一样回退默认场景
	}
	console.warn('[mock] 场景数据不合法，已使用默认场景')
	return DEFAULT_SCENARIO
}

function isScenario(value: unknown): value is Scenario {
	const v = value as Partial<Scenario> | null
	return (
		typeof v?.loggedIn === 'boolean' &&
		REWARD_STATES.includes(v.rewardState as RewardState) &&
		SWITCHES.every((key) => typeof v.switches?.[key] === 'boolean') &&
		(v.popup === null || POPUPS.includes(v.popup as PopupKey)) &&
		typeof v.latencyMs === 'number' &&
		v.latencyMs >= 0
	)
}

import axios from 'axios'
import type { AxiosAdapter, AxiosResponse, InternalAxiosRequestConfig } from 'axios'
import { writeState } from './state'
import type { MockContext, MockEnvelope, MockHandler, MockRoutes, Scenario, SessionState } from './types'

/** 需要从路径中去掉的接口前缀，按顺序匹配 */
const PATH_PREFIXES = ['/api/webapi', '/api']

/**
 * 从完整请求地址提取接口路径。
 *
 * @param url - 请求地址，可以带域名和查询串。
 * @returns 去掉域名、查询串和 `/api/webapi`、`/api` 前缀后的路径，如 `/GetActivityList`。
 */
export function toMockPath(url: string): string {
	const { pathname } = new URL(url, location.href)
	const prefix = PATH_PREFIXES.find((item) => pathname.startsWith(item + '/'))
	return prefix ? pathname.slice(prefix.length) : pathname
}

/**
 * 创建按接口路径分发的 axios adapter。
 *
 * @remarks 所有请求都在本地完成，不发起网络请求，也不响应取消信号：被取消的重复请求同样会拿到数据。
 * 未命中路由表的接口返回兜底数据，并去重记入 `window.__protoMock.unhandled`。
 * 响应数据每次深拷贝，页面修改返回值不会污染假数据常量。
 * @param routes - 路由表。
 * @param scenario - 本次启动的场景。
 * @param state - 会话状态；处理函数修改后由 adapter 持久化。
 * @returns 可以赋给 `axios.defaults.adapter` 的 adapter。
 */
export function createMockAdapter(routes: MockRoutes, scenario: Scenario, state: SessionState): AxiosAdapter {
	return async (config) => {
		const url = axios.getUri(config)
		const path = toMockPath(url)
		await new Promise((resolve) => setTimeout(resolve, scenario.latencyMs))
		const handler = routes[path]
		if (!handler) return respond(config, fallback(path))
		return respond(config, run(handler, { path, body: readBody(url, config.data), scenario, state }))
	}
}

function run(handler: MockHandler, ctx: MockContext): MockEnvelope {
	try {
		const envelope = handler(ctx)
		writeState(ctx.state)
		return envelope
	} catch (error) {
		console.error('[mock] 处理函数异常: ' + ctx.path, error)
		return { code: -1, msg: 'System Error', data: null }
	}
}

function fallback(path: string): unknown {
	const { unhandled } = window.__protoMock
	if (!unhandled.includes(path)) {
		unhandled.push(path)
		console.warn('[mock] 未覆盖: ' + path)
	}
	// ArUPI 支付接口没有信封，开奖数据是静态 JSON，两者都返回空对象
	if (path.endsWith('.json') || path.startsWith('/ar-wallet/')) return {}
	return { code: 0, msg: 'Succeed', data: { list: [], totalCount: 0, totalPage: 1 } }
}

function readBody(url: string, data: unknown): Record<string, unknown> {
	const query = Object.fromEntries(new URL(url, location.href).searchParams)
	if (typeof data !== 'string') return query
	try {
		const body: unknown = JSON.parse(data)
		return body && typeof body === 'object' ? { ...query, ...(body as Record<string, unknown>) } : query
	} catch {
		return query
	}
}

function respond(config: InternalAxiosRequestConfig, data: unknown): AxiosResponse {
	return { data: structuredClone(data), status: 200, statusText: 'OK', headers: {}, config, request: {} }
}

	if (item.kind === 'none') return

		phone.src = item.target
		return

	const hadPopup = Boolean(scenario && scenario.popup)
	if (hadPopup) saveScenario({ ...scenario, popup: null })
	openH5(item.target, hadPopup)
}
